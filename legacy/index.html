📊 Cara Menghubungkan ke Google Spreadsheet

  1. Buka Google Spreadsheet baru di akun Google kantor Anda.
  2. Klik menu Ekstensi (Extensions) → Apps Script.
  3. Hapus kode yang ada, lalu Salin (Copy) kode di bawah ini dan tempelkan:
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var payload = JSON.parse(e.postData.contents);
  
  // 1. Simpan Log setiap perubahan ke Sheet "Log_Proses"
  var sheetLog = ss.getSheetByName("Log_Proses") || ss.insertSheet("Log_Proses");
  if (sheetLog.getLastRow() === 0) {
    sheetLog.appendRow(["Waktu", "Kategori", "Tahapan", "Pelaku", "Unit", "Rincian"]);
  }
  if (payload.logTerbaru) {
    var l = payload.logTerbaru;
    sheetLog.appendRow([l.waktu, l.kategori, l.judul, l.pelaku, l.unit, l.rincian]);
  }

  // 2. Simpan Snapshot Database penuh ke Sheet "Backup_JSON"
  var sheetDb = ss.getSheetByName("Backup_JSON") || ss.insertSheet("Backup_JSON");
  sheetDb.clear();
  sheetDb.getRange(1, 1).setValue("Waktu Sinkronisasi Terakhir: " + new Date().toLocaleString("id-ID"));
  sheetDb.getRange(2, 1).setValue(JSON.stringify(payload.database));

  return ContentService.createTextOutput(JSON.stringify({ status: "OK" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetDb = ss.getSheetByName("Backup_JSON");
  var raw = sheetDb ? sheetDb.getRange(2, 1).getValue() : "{}";
  return ContentService.createTextOutput(raw)
    .setMimeType(ContentService.MimeType.JSON);
}
  1. Klik tombol Deploy → New deployment (Deployment baru) di kanan atas Apps Script.
  2. Pilih jenis Web app, isi Execute as: Me, dan Who has access: Anyone (Siapa saja), lalu klik Deploy.
  3. Salin Web App URL (`https://script.google.com/macros/s/.../exec`) dan tempelkan pada kolom URL Web App Google Spreadsheet di aplikasi ini.
